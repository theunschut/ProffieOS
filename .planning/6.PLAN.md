**Step 4: Implement RtTrJoin (5+ uses)**
TrJoin runs multiple transitions in parallel, blending their progress.
```cpp
// TrJoin<Tr1, Tr2, ...>: run transitions in parallel
class RtTrJoin : public RtTransNode {
public:
  RtTrJoin(std::initializer_list<RtTransNode*> transitions)
    : transitions_(transitions.begin(), transitions.end()) {}
  void run(BladeBase* blade) override {
    for (auto tr : transitions_) tr->run(blade);
  }
  int getProgress(BladeBase* blade) override {
    // Average progress across all transitions
    if (transitions_.empty()) return 0;
    int total = 0;
    for (auto tr : transitions_) total += tr->getProgress(blade);
    return total / transitions_.size();
  }
private:
  RtVec<RtTransNode*> transitions_;
};
```

    }

    if (styles_.empty()) return RGBA_um::Transparent();
    return styles_[current_style_]->getColor(led);

    current_style_ = (current_style_ + 1) % styles_.size();

  int current_style_;
